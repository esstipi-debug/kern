```python
"""Flow map engine -- the observed supply network, measured.

Pure and dependency-light (stdlib + scipy's normal quantile via
``src.safety_stock.service_level_factor``): it takes movements the client
actually recorded and returns the network they imply, the transit statistics of
every leg, and the capital that each leg's variability forces downstream.

Deliberately I/O-free, like the other analytical cores: ``jobs/flow_map_job.py``
owns file reading, the QA gate and the deliverables.

An edge with too few observations is reported as UNMEASURABLE with empty
statistics -- never as zero. A declared hole is an honest result; an assumed
transit time is not.
"""

from __future__ import annotations

from collections import defaultdict
from dataclasses import dataclass, field

MEASURED = "measured"
UNMEASURABLE = "unmeasurable"


@dataclass(frozen=True)
class Movement:
    """One observed shipment leg. ``transit_days`` may be negative (bad data).

    ``departed_day`` is the departure date as a day ordinal. It exists so the
    calendar span the movements cover is derivable from the movements themselves,
    with no side channel between the job's prepare and run stages.
    """

    origin: str
    destination: str
    product: str
    quantity: float
    transit_days: float
    unit_cost: float = 0.0
    departed_day: float = 0.0


@dataclass(frozen=True)
class FlowGraph:
    """The network implied by the movements. ``echelon`` is depth from a source."""

    nodes: tuple[str, ...]
    echelon: dict[str, int] = field(default_factory=dict)
    edges: tuple[tuple[str, str], ...] = ()
    bidirectional: tuple[tuple[str, str], ...] = ()
    cycle_note: str = ""


def _edge_volume(movements: list[Movement]) -> dict[tuple[str, str], float]:
    volume: dict[tuple[str, str], float] = defaultdict(float)
    for m in movements:
        volume[(m.origin, m.destination)] += m.quantity
    return dict(volume)


def _bidirectional_pairs(
    volume: dict[tuple[str, str], float],
) -> tuple[tuple[tuple[str, str], ...], set[tuple[str, str]]]:
    """Find A<->B pairs; return the pairs (sorted, canonical) and the weaker legs."""
    pairs: list[tuple[str, str]] = []
    weaker: set[tuple[str, str]] = set()
    for (a, b) in sorted(volume):
        if a >= b or (b, a) not in volume:
            continue
        pairs.append((a, b))
        weaker.add((a, b) if volume[(a, b)] <= volume[(b, a)] else (b, a))
    return tuple(pairs), weaker


def build_graph(movements: list[Movement]) -> FlowGraph:
    """Nodes, sorted edges, topological echelons, and an explicit cycle note."""
    volume = _edge_volume(movements)
    edges = tuple(sorted(volume))
    nodes_set = {n for edge in edges for n in edge}
    pairs, weaker = _bidirectional_pairs(volume)

    acyclic = [e for e in edges if e not in weaker]
    indegree = {n: 0 for n in nodes_set}
    downstream: dict[str, list[str]] = defaultdict(list)
    for origin, destination in acyclic:
        indegree[destination] += 1
        downstream[origin].append(destination)

    sources = sorted(n for n in nodes_set if indegree[n] == 0)
    notes: list[str] = []
    if pairs:
        legs = ", ".join(f"{a} <-> {b}" for a, b in pairs)
        notes.append(
            f"bidirectional flow detected ({legs}); the lower-volume leg is ignored "
            "when computing echelons"
        )

    echelon = {n: 0 for n in nodes_set}
    if not sources:
        notes.append("no source node (every node has inbound flow); echelons flattened to 0")
    else:
        frontier = list(sources)
        seen = set(sources)
        depth = 0
        while frontier:
            depth += 1
            nxt: list[str] = []
            for node in frontier:
                for child in sorted(downstream[node]):
                    if child in seen:
                        continue
                    echelon[child] = depth
                    seen.add(child)
                    nxt.append(child)
            frontier = sorted(nxt)

    nodes = tuple(sorted(nodes_set, key=lambda n: (echelon[n], n)))
    return FlowGraph(
        nodes=nodes,
        echelon=echelon,
        edges=edges,
        bidirectional=pairs,
        cycle_note="; ".join(notes),
    )
```

```python
MIN_OBSERVATIONS = 8  # below this an edge reports no transit statistics at all


@dataclass(frozen=True)
class EdgeStat:
    """One leg's observed transit behaviour. Statistics are None when UNMEASURABLE."""

    origin: str
    destination: str
    n: int
    volume: float
    status: str
    mean_days: float | None = None
    stdev_days: float | None = None
    p50_days: float | None = None
    p90_days: float | None = None
    mean_unit_cost: float = 0.0
    bidirectional: bool = False


def _percentile(values: list[float], q: float) -> float:
    """Linear-interpolation percentile on a sorted list (numpy's default method)."""
    ordered = sorted(values)
    if len(ordered) == 1:
        return ordered[0]
    position = (len(ordered) - 1) * q
    low = int(position)
    high = min(low + 1, len(ordered) - 1)
    return ordered[low] + (ordered[high] - ordered[low]) * (position - low)


def _stdev(values: list[float], mean: float) -> float:
    """Sample standard deviation (n-1). Callers guarantee len(values) >= 2."""
    variance = sum((v - mean) ** 2 for v in values) / (len(values) - 1)
    return variance**0.5


def measure_edges(
    movements: list[Movement],
    *,
    min_observations: int = MIN_OBSERVATIONS,
    bidirectional: tuple[tuple[str, str], ...] = (),
) -> tuple[EdgeStat, ...]:
    """Transit statistics per edge, sorted by (origin, destination)."""
    grouped: dict[tuple[str, str], list[Movement]] = defaultdict(list)
    for m in movements:
        grouped[(m.origin, m.destination)].append(m)

    two_way = {node for pair in bidirectional for node in pair}
    stats: list[EdgeStat] = []
    for (origin, destination) in sorted(grouped):
        rows = grouped[(origin, destination)]
        days = [r.transit_days for r in rows]
        volume = sum(r.quantity for r in rows)
        cost_rows = [r for r in rows if r.unit_cost > 0]
        mean_cost = sum(r.unit_cost for r in cost_rows) / len(cost_rows) if cost_rows else 0.0
        is_two_way = origin in two_way and destination in two_way
        if len(days) < min_observations or len(days) < 2:
            stats.append(EdgeStat(
                origin=origin, destination=destination, n=len(days), volume=volume,
                status=UNMEASURABLE, mean_unit_cost=mean_cost, bidirectional=is_two_way,
            ))
            continue
        mean = sum(days) / len(days)
        stats.append(EdgeStat(
            origin=origin, destination=destination, n=len(days), volume=volume,
            status=MEASURED, mean_days=mean, stdev_days=_stdev(days, mean),
            p50_days=_percentile(days, 0.5), p90_days=_percentile(days, 0.9),
            mean_unit_cost=mean_cost, bidirectional=is_two_way,
        ))
    return tuple(stats)
```

```python
_TOP_OPTIONS = 3


@dataclass(frozen=True)
class EdgeBuffer:
    """Protective inventory an edge's transit variability forces downstream."""

    origin: str
    destination: str
    buffer_units: float
    buffer_value: float


@dataclass(frozen=True)
class FlowReport:
    """Everything the job layer needs: the network, the measurement, the verdict."""

    graph: FlowGraph
    edges: tuple[EdgeStat, ...]
    buffers: tuple[EdgeBuffer, ...]
    n_movements: int
    n_negative_transit: int
    unmeasurable_volume_share: float
    span_days: float
    outcome: GuidedOutcome
    summary: str


def _buffers(
    edges: tuple[EdgeStat, ...], *, z: float, span_days: float
) -> tuple[EdgeBuffer, ...]:
    """z * mean_daily_demand * sigma_LT, valued at the edge's mean unit cost.

    Only the lead-time-variability term of the combined safety-stock formula:
    ``src.safety_stock.safety_stock`` computes the DEMAND-variability term, which
    an edge's transit noise does not drive. Only ``service_level_factor`` is reused.
    """
    out: list[EdgeBuffer] = []
    for e in edges:
        if e.status != MEASURED or e.stdev_days is None:
            continue
        units = z * (e.volume / span_days) * e.stdev_days
        out.append(EdgeBuffer(
            origin=e.origin, destination=e.destination,
            buffer_units=units, buffer_value=units * e.mean_unit_cost,
        ))
    return tuple(sorted(
        out, key=lambda b: (-b.buffer_value, -b.buffer_units, b.origin, b.destination)
    ))


def _options(buffers: tuple[EdgeBuffer, ...], edges: tuple[EdgeStat, ...]) -> GuidedOutcome:
    by_edge = {(e.origin, e.destination): e for e in edges}
    options: list[ExecutionOption] = []
    for rank, b in enumerate(buffers[:_TOP_OPTIONS]):
        e = by_edge[(b.origin, b.destination)]
        options.append(ExecutionOption(
            label=f"Stabilise {b.origin} -> {b.destination}",
            summary=(
                f"Transit averages {e.mean_days:.1f} d with a {e.stdev_days:.1f} d spread "
                f"over {e.n} observed movement(s); that spread alone forces "
                f"{b.buffer_units:,.0f} unit(s) of protective stock "
                f"({b.buffer_value:,.0f} in value) downstream."
            ),
            score=b.buffer_value,
            recommended=rank == 0,
            action=f"review_lane:{b.origin}->{b.destination}",
            tradeoffs=(
                "Cutting transit variability on this lane releases the buffer; cutting the "
                "average transit time does not, on its own, release any of it."
            ),
        ))
    residual = Residual(
        description=(
            "Why the lane varies (carrier, customs, dock congestion, batching) is not in the "
            "movement data; confirm the cause with the lane owner before acting."
        ),
        risk_if_skipped="Acting on the wrong root cause leaves the buffer in place.",
    )
    return as_options(options, residuals=[residual])


def analyze(
    movements: list[Movement],
    *,
    min_observations: int = MIN_OBSERVATIONS,
    cycle_service_level: float = 0.95,
    span_days: float = 0.0,
) -> FlowReport:
    """Build the graph, measure every edge, and rank the capital each lane locks up."""
    if span_days <= 0:
        raise ValueError("span_days must be > 0 (the calendar span the movements cover)")
    graph = build_graph(movements)
    edges = measure_edges(
        movements, min_observations=min_observations, bidirectional=graph.bidirectional
    )
    total_volume = sum(e.volume for e in edges)
    unmeasurable_volume = sum(e.volume for e in edges if e.status == UNMEASURABLE)
    share = unmeasurable_volume / total_volume if total_volume > 0 else 0.0
    buffers = _buffers(
        edges, z=service_level_factor(cycle_service_level), span_days=span_days
    )
    outcome = _options(buffers, edges)
    n_measured = sum(1 for e in edges if e.status == MEASURED)
    top = (
        f"the {buffers[0].origin} -> {buffers[0].destination} lane locks up the most capital"
        if buffers else "no lane has enough observations to rank"
    )
    summary = (
        f"Mapped {len(graph.nodes)} node(s) and {len(edges)} lane(s) from "
        f"{len(movements)} movement(s); {n_measured} lane(s) measurable, "
        f"{share * 100:.0f}% of volume on unmeasurable lanes; {top}."
    )
    return FlowReport(
        graph=graph, edges=edges, buffers=buffers, n_movements=len(movements),
        n_negative_transit=sum(1 for m in movements if m.transit_days < 0),
```

```python
"""End-to-end flow map job -- intake, QA gate, deliverables.

Reads a movements export (one row per shipment leg) with pandas directly, the
same way ``jobs/disruption_scan_job.py`` reads a supplier list: the canonical
demand intake in ``jobs/intake.py`` is shaped for date/product/quantity series,
not for origin-destination legs.

The gate refuses to dress up thin data: if most volume rides on lanes with too
few observations, or the verdict leans on one of them, nothing ships.
"""

from __future__ import annotations

import re
from pathlib import Path

import pandas as pd

from src.export import write_summary_csv
from src.flow_map import (
    MEASURED,
    MIN_OBSERVATIONS,
    UNMEASURABLE,
    FlowReport,
    Movement,
    analyze,
)
from src.guided import verify_guided

MAX_UNMEASURABLE_SHARE = 0.40
MAX_NEGATIVE_TRANSIT_SHARE = 0.01
MIN_EDGES = 2

COLUMN_ALIASES: dict[str, tuple[str, ...]] = {
    "fecha_salida": ("fechasalida", "shipdate", "departuredate", "fechadespacho", "dispatchdate"),
    "fecha_llegada": ("fechallegada", "receiptdate", "arrivaldate", "fecharecepcion", "deliverydate"),
    "origen": ("origen", "origin", "from", "fromlocation", "source", "desde"),
    "destino": ("destino", "destination", "to", "tolocation", "hacia"),
    "producto": ("producto", "product", "productid", "sku", "item"),
    "cantidad": ("cantidad", "quantity", "qty", "units", "volume"),
    "costo_unitario": ("costounitario", "unitcost", "cost", "unitprice", "price"),
}
_REQUIRED = ("fecha_salida", "fecha_llegada", "origen", "destino", "producto", "cantidad")


def _norm(header: str) -> str:
    return re.sub(r"[^a-z0-9]", "", str(header).lower())


def _resolve_columns(df: pd.DataFrame) -> dict[str, str]:
    """Canonical field -> the client's own column name. Raises naming what is missing."""
    found: dict[str, str] = {}
    for header in df.columns:
        norm = _norm(header)
        for canonical, aliases in COLUMN_ALIASES.items():
            if canonical not in found and norm in aliases:
                found[canonical] = header
                break
    missing = [c for c in _REQUIRED if c not in found]
    if missing:
        raise ValueError(
            "missing required column(s): " + ", ".join(missing)
            + ". A movements export needs one row per shipment leg with a departure date, "
            "an arrival date, an origin, a destination, a product and a quantity."
        )
    return found


def prepare(data_path: str, params: dict | None = None) -> list[Movement]:
    """Load the movements file and turn each row into a Movement with measured transit."""
    path = Path(data_path)
    if not path.exists():
        raise FileNotFoundError(f"{data_path} does not exist")
    df = pd.read_excel(path) if path.suffix.lower() in {".xlsx", ".xls"} else pd.read_csv(path)
    if df.empty:
        raise ValueError("the movements file has no rows")
    cols = _resolve_columns(df)

    departed = pd.to_datetime(df[cols["fecha_salida"]], errors="coerce")
    arrived = pd.to_datetime(df[cols["fecha_llegada"]], errors="coerce")
    cost_col = cols.get("costo_unitario")

    movements: list[Movement] = []
    for i in range(len(df)):
        if pd.isna(departed.iloc[i]) or pd.isna(arrived.iloc[i]):
            continue
        quantity = pd.to_numeric(df[cols["cantidad"]].iloc[i], errors="coerce")
        if pd.isna(quantity):
            continue
        cost = 0.0
        if cost_col is not None:
            parsed = pd.to_numeric(df[cost_col].iloc[i], errors="coerce")
            cost = 0.0 if pd.isna(parsed) else float(parsed)
        movements.append(Movement(
            origin=str(df[cols["origen"]].iloc[i]).strip(),
            destination=str(df[cols["destino"]].iloc[i]).strip(),
            product=str(df[cols["producto"]].iloc[i]).strip(),
            quantity=float(quantity),
            transit_days=(arrived.iloc[i] - departed.iloc[i]).total_seconds() / 86400.0,
            unit_cost=cost,
            departed_day=departed.iloc[i].toordinal(),
        ))
    if not movements:
        raise ValueError("no usable movements: every row was missing a date or a quantity")
    return movements


def calendar_span(movements: list[Movement]) -> float:
    """Days from the first departure to the last arrival, floored at 1."""
    first = min(m.departed_day for m in movements)
    last = max(m.departed_day + m.transit_days for m in movements)
    return max(last - first, 1.0)


def run(movements: list[Movement], params: dict | None = None) -> FlowReport:
    """Analyse the movements. The calendar span is derived from the movements."""
    params = params or {}
    span = float(params.get("span_days") or calendar_span(movements))
    return analyze(
        movements,
        min_observations=int(params.get("min_observations", MIN_OBSERVATIONS)),
        cycle_service_level=float(params.get("service_level", 0.95)),
        span_days=span,
    )


def verify(report: FlowReport) -> list[str]:
    """QA gate: four conditions. A gate that thin data can pass is not a gate."""
    issues = list(verify_guided(report.outcome))
    if report.unmeasurable_volume_share > MAX_UNMEASURABLE_SHARE:
        issues.append(
            f"{report.unmeasurable_volume_share * 100:.0f}% of volume rides on unmeasurable "
            f"lanes (limit {MAX_UNMEASURABLE_SHARE * 100:.0f}%)"
        )
    if report.n_movements and (
        report.n_negative_transit / report.n_movements > MAX_NEGATIVE_TRANSIT_SHARE
    ):
        issues.append(
            f"{report.n_negative_transit} movement(s) record arrival before departure "
            f"(limit {MAX_NEGATIVE_TRANSIT_SHARE * 100:.0f}% of rows)"
        )
    if len(report.edges) < MIN_EDGES:
        issues.append(f"the graph has fewer than two lanes ({len(report.edges)})")
    ranked = {(b.origin, b.destination) for b in report.buffers[:3]}
    unmeasurable = {
        (e.origin, e.destination) for e in report.edges if e.status == UNMEASURABLE
    }
    if ranked & unmeasurable:
        issues.append("a ranked option leans on an unmeasurable lane")
    return issues
```

```python
"""Deterministic sankey rendering for a FlowReport -- no dependencies, no CDN.

The deliverable has to open on a plant PC with no internet, travel by mail and
survive an air-gapped client, so the layout is computed here rather than handed
to a JS library. Everything is sorted explicitly: same report in, byte-identical
SVG out, which is what makes the golden test meaningful.
"""

from __future__ import annotations

from html import escape

from src.flow_map import MEASURED, UNMEASURABLE, EdgeStat, FlowReport

_COL_WIDTH = 220
_NODE_WIDTH = 18
_NODE_GAP = 14
_MIN_NODE_HEIGHT = 8
_MARGIN = 40
_MAX_STROKE = 26.0
_MIN_STROKE = 1.5


def _node_throughput(report: FlowReport) -> dict[str, float]:
    flow: dict[str, float] = {n: 0.0 for n in report.graph.nodes}
    for e in report.edges:
        flow[e.origin] = flow.get(e.origin, 0.0) + e.volume
        flow[e.destination] = flow.get(e.destination, 0.0) + e.volume
    return flow


def _layout(report: FlowReport) -> tuple[dict[str, tuple[float, float, float]], int, int]:
    """node -> (x, y, height), plus canvas width and height. Deterministic by sort."""
    echelon = report.graph.echelon
    columns: dict[int, list[str]] = {}
    for node in report.graph.nodes:
        columns.setdefault(echelon[node], []).append(node)

    throughput = _node_throughput(report)
    tallest_column = max(
        (sum(throughput[n] for n in nodes) for nodes in columns.values()), default=1.0
    ) or 1.0
    usable_height = 520.0

    placed: dict[str, tuple[float, float, float]] = {}
    for level in sorted(columns):
        y = float(_MARGIN)
        for node in sorted(columns[level]):
            height = max(
                _MIN_NODE_HEIGHT, usable_height * (throughput[node] / tallest_column)
            )
            placed[node] = (float(_MARGIN + level * _COL_WIDTH), y, height)
            y += height + _NODE_GAP
    width = _MARGIN * 2 + (max(columns) * _COL_WIDTH) + _NODE_WIDTH
    height = int(max(y for _, y, _ in placed.values()) + _MARGIN + 40)
    return placed, int(width), height


def _stroke_width(edge: EdgeStat, max_volume: float) -> float:
    if max_volume <= 0:
        return _MIN_STROKE
    return _MIN_STROKE + (_MAX_STROKE - _MIN_STROKE) * (edge.volume / max_volume)


def _edge_title(edge: EdgeStat) -> str:
    if edge.status == MEASURED:
        return (
            f"{edge.origin} -> {edge.destination}: {edge.volume:,.0f} units, "
            f"transit {edge.mean_days:.1f} d +/- {edge.stdev_days:.1f} d over {edge.n} movements"
        )
    return (
        f"{edge.origin} -> {edge.destination}: {edge.volume:,.0f} units, "
        f"transit not measurable ({edge.n} movement(s), below the threshold)"
    )


def render_sankey(report: FlowReport, *, client: str = "Client", width: int = 960) -> str:
    """One SVG string. `width` is advisory: the real width comes from the layout."""
    placed, canvas_width, canvas_height = _layout(report)
    canvas_width = max(canvas_width, width)
    max_volume = max((e.volume for e in report.edges), default=0.0)

    parts = [
        f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 {canvas_width} {canvas_height}" '
        f'width="100%" role="img">',
        f"<title>{escape(client)} flow map</title>",
        "<desc>Observed supply network by echelon; band width is volume, dashed lanes "
        "have too few movements to measure transit.</desc>",
    ]
    for edge in report.edges:
        if edge.origin not in placed or edge.destination not in placed:
            continue
        x1, y1, h1 = placed[edge.origin]
        x2, y2, h2 = placed[edge.destination]
        sx, sy = x1 + _NODE_WIDTH, y1 + h1 / 2
        ex, ey = x2, y2 + h2 / 2
        mid = (sx + ex) / 2
        dash = ' stroke-dasharray="6 5"' if edge.status == UNMEASURABLE else ""
        parts.append(
            f'<path d="M{sx:.1f},{sy:.1f} C{mid:.1f},{sy:.1f} {mid:.1f},{ey:.1f} '
            f'{ex:.1f},{ey:.1f}" fill="none" stroke="#5DCAA5" stroke-opacity="0.55" '
            f'stroke-width="{_stroke_width(edge, max_volume):.1f}"{dash}>'
            f"<title>{escape(_edge_title(edge))}</title></path>"
        )
    for node in sorted(placed):
        x, y, h = placed[node]
        parts.append(
            f'<rect x="{x:.1f}" y="{y:.1f}" width="{_NODE_WIDTH}" height="{h:.1f}" '
            f'rx="3" fill="#0F6E56"><title>{escape(node)}</title></rect>'
        )
        parts.append(
            f'<text x="{x + _NODE_WIDTH + 8:.1f}" y="{y + h / 2 + 4:.1f}" '
            f'font-family="system-ui, sans-serif" font-size="13" fill="#2C2C2A">'
            f"{escape(node)}</text>"
        )
    parts.append("</svg>")
    return "".join(parts)


def render_page(report: FlowReport, *, client: str = "Client") -> str:
    """A single self-contained HTML file: the map, the note, the unmeasurable legend."""
    unmeasurable = [e for e in report.edges if e.status == UNMEASURABLE]
    note = escape(report.graph.cycle_note) if report.graph.cycle_note else ""
    rows = "".join(
        f"<li>{escape(e.origin)} -> {escape(e.destination)}: {e.n} movement(s)</li>"
        for e in unmeasurable
    )
    return (
        "<!doctype html><html lang=\"en\"><head><meta charset=\"utf-8\">"
        f"<title>{escape(client)} flow map</title>"
        "<style>body{font-family:system-ui,sans-serif;margin:2rem;color:#2C2C2A;"
        "background:#fff;max-width:1100px}h1{font-weight:500;font-size:22px}"
        "p,li{font-size:14px;line-height:1.6}</style></head><body>"
        f"<h1>{escape(client)} - observed flow map</h1>"
        f"<p>{escape(report.summary)}</p>"
        + render_sankey(report, client=client)
        + (f"<p>{note}</p>" if note else "")
        + (
            "<p>Lanes with too few movements to measure transit (drawn dashed, excluded "
            f"from the ranking):</p><ul>{rows}</ul>" if unmeasurable else ""
        )
        + "</body></html>"
    )
```

```python
def write_operational(
    report: FlowReport, out_dir: str | Path, client: str = "Client"
) -> dict[str, Path]:

    from src.flow_svg import render_page

    d = Path(out_dir)

    for e in report.edges:
        b = buffer_by_edge.get((e.origin, e.destination))
        rows.append({
            "origin": e.origin,
            "destination": e.destination,
            "echelon_origin": report.graph.echelon.get(e.origin, ""),
            "echelon_destination": report.graph.echelon.get(e.destination, ""),
            "movements": e.n,
            "volume": round(e.volume, 2),
            "status": e.status,
            "mean_days": "" if e.mean_days is None else round(e.mean_days, 2),
            "stdev_days": "" if e.stdev_days is None else round(e.stdev_days, 2),
            "p50_days": "" if e.p50_days is None else round(e.p50_days, 2),
            "p90_days": "" if e.p90_days is None else round(e.p90_days, 2),
            "buffer_units": "" if b is None else round(b.buffer_units, 2),
            "buffer_value": "" if b is None else round(b.buffer_value, 2),
            "bidirectional": "yes" if e.bidirectional else "",
        })
    html_path = d / "flow_map.html"
```

```python
def build_deck(
    report: FlowReport,
    *,
    client: str = "Client",
    prepared: str = "",
    citations: tuple[str, ...] = (),
    confidence: float = 0.6,
    lang: str = "en",
) -> Deliverable:
    """The client-facing map: which lane locks up capital, and what we could not measure."""
    findings: list[Finding] = []
    if report.buffers:
        top = report.buffers[0]
        edge = next(
            e for e in report.edges
            if (e.origin, e.destination) == (top.origin, top.destination)
        )
        findings.append(Finding(
            f"Transit variability on {top.origin} -> {top.destination} locks up the most capital",
            f"Measured over {edge.n} movement(s): {edge.mean_days:.1f} d average transit, "
            f"{edge.stdev_days:.1f} d standard deviation, p90 {edge.p90_days:.1f} d. That spread "
            f"forces roughly {top.buffer_units:,.0f} unit(s) of protective stock downstream.",
            impact="stabilising this lane releases the buffer it forces",
        ))
        for b in report.buffers[1:3]:
            findings.append(Finding(
                f"Next lane: {b.origin} -> {b.destination}",
                f"Forces roughly {b.buffer_units:,.0f} unit(s) of protective stock.",
                impact="queue behind the top lane",
            ))
    else:
        findings.append(Finding(
            "No lane has enough observations to measure",
            f"Every one of the {len(report.edges)} lane(s) fell below the "
            f"{MIN_OBSERVATIONS}-movement threshold, so no transit statistic is reported.",
            impact="widen the export window before treating any lane as stable",
        ))

    unmeasurable = [e for e in report.edges if e.status == UNMEASURABLE]
    if unmeasurable:
        names = ", ".join(f"{e.origin} -> {e.destination}" for e in unmeasurable[:5])
        findings.append(Finding(
            "Lanes we could not measure",
            f"{len(unmeasurable)} lane(s) carry fewer than {MIN_OBSERVATIONS} observed "
            f"movement(s) ({names}). They are drawn on the map but report no transit "
            "number and are excluded from the ranking.",
            impact="an empty cell is a declared gap, not a zero",
        ))
    if report.graph.cycle_note:
        findings.append(Finding(
            "Bidirectional flow in the network",
            report.graph.cycle_note,
            impact="confirm whether these legs are intended or are re-handling",
        ))

    measured = sum(1 for e in report.edges if e.status == MEASURED)
    kpis = (
        Kpi("Nodes mapped", f"{len(report.graph.nodes)}", rationale="Distinct origins and destinations"),
        Kpi("Lanes mapped", f"{len(report.edges)}", rationale="Distinct origin-destination pairs"),
        Kpi("Lanes measurable", f"{measured}", target=f"{len(report.edges)}",
            rationale=f"Lanes with at least {MIN_OBSERVATIONS} observed movements"),
        Kpi("Volume on unmeasurable lanes",
            f"{report.unmeasurable_volume_share * 100:.0f}%", target="0%",
            rationale="Share of volume with no transit statistic"),
        Kpi("Movements read", f"{report.n_movements}", rationale="Rows that carried both dates and a quantity"),
    )
    data_sources = (
        DataSource("Movements export (origin / destination / dates / quantity)", "client-supplied", "per run"),
    )
    recommendations = [
        "Stabilise the top-ranked lane before adding stock to absorb it.",
        "Widen the export window for any lane reported as unmeasurable, then re-run.",
        "Feed the measured per-lane transit spread into the safety-stock review; it replaces an assumption.",
    ]
    return Deliverable(
        title="Flow Map - Observed Network and Measured Transit",
        client=client,
        summary=report.summary,
        findings=tuple(findings),
        kpis=kpis,
        data_sources=data_sources,
        recommendations=tuple(recommendations),
        options=tuple(report.outcome.options),
        citations=tuple(citations),
        confidence=confidence,
        residual=(
            "Transit is measured from the movement dates the client recorded, not from carrier "
            "telemetry: a late-keyed receipt inflates the spread. The map shows the network that "
            "actually moved goods in the export window, which may omit seasonal lanes."
        ),
        prepared=prepared,
        lang=lang,
    )
```
